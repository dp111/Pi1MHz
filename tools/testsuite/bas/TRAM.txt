10 REM TRAM - RAM expansion: page mode, byte mode, 64 KB burst at 6502 speed
20 REM Avoids JIM &3000-&5FFF (M5000 registers) and the top of JIM (services)
30 PROCbegin("RAM")
40 IF PAGE=&800 PROCskip("all","tube: FRED/JIM not addressable"):PROCend
50 DIM G%(11):G%(0)=1:G%(1)=2:G%(2)=255:G%(3)=256:G%(4)=4095:G%(5)=4096:G%(6)=&8000:G%(7)=&10000:G%(8)=&1FFFF:G%(9)=&2FFFF:G%(10)=&3FFFF:G%(11)=&12345
60 FOR J%=0 TO 11:PROCpage(G%(J%)):FOR I%=0 TO 7:?(&FD00+I%)=(G%(J%)+I%) AND 255:?(&FDF8+I%)=(G%(J%) DIV 256+I%*3) AND 255:NEXT:NEXT
70 e%=0:FOR J%=0 TO 11:PROCpage(G%(J%)):FOR I%=0 TO 7:IF ?(&FD00+I%)<>((G%(J%)+I%) AND 255) OR ?(&FDF8+I%)<>((G%(J%) DIV 256+I%*3) AND 255) e%=e%+1
80 NEXT:NEXT:PROCinfo("page errs",STR$e%):PROCchk("page mode 12 pages to 64MB",e%=0)
90 PROCpage(&77):FOR I%=0 TO 255:?(&FD00+I%)=(I%*7) AND 255:NEXT:e%=0:FOR I%=0 TO 255:IF ?(&FD00+I%)<>((I%*7) AND 255) e%=e%+1
100 NEXT:PROCchk("full page fill",e%=0)
105 REM byte mode (Sprow 16 MB) is the LAST 16 MB of JIM: page (size-1)*&10000 + A DIV 256
106 ?&FCCA=0:s%=?&FCCB:PROCinfo("jim size 16MB units",STR$s%)
110 IF s%<2 PROCskip("byte mode","no JIM size from &FCCA/&FCCB") ELSE b%=(s%-1)*&10000:?&FC00=&56:?&FC01=&34:?&FC02=&12:?&FC03=&5A:PROCpage(b%+&1234):PROCchk("byte mode write, page read",?&FD56=&5A)
120 IF s%>=2 ?&FD57=&C3:?&FC00=&57:?&FC01=&34:?&FC02=&12:PROCchk("page write, byte mode read",?&FC03=&C3)
130 DIM C% 300:FOR O%=0 TO 2 STEP 2:P%=C%:[OPT O%
140 .wr LDA #8:STA &FCFE:LDA #0:STA &FCFD
150 .wrp STA &FCFF:STA &72:LDX #0
160 .wrb TXA:EOR &72:STA &FD00,X:INX:BNE wrb
170 LDA &72:CLC:ADC #1:BNE wrp:RTS
180 .rd LDA #8:STA &FCFE:LDA #0:STA &FCFD:STA &70:STA &71
190 .rdp STA &FCFF:STA &72:LDX #0
200 .rdb TXA:EOR &72:CMP &FD00,X:BEQ rok:INC &70:BNE rok:INC &71
210 .rok INX:BNE rdb:LDA &72:CLC:ADC #1:BNE rdp:RTS
220 .coh LDA #0:STA &70:STA &71:STA &FCFD:STA &FCFF:LDA #9:STA &FCFE:LDY #64
230 .cop STY &73:LDX #0
240 .cob TXA:EOR &73:STA &FD00,X:CMP &FD00,X:BEQ cok:INC &70:BNE cok:INC &71
250 .cok INX:BNE cob:DEY:BNE cop:RTS
260 ]:NEXT
270 T%=TIME:CALL wr:t1%=TIME-T%:T%=TIME:CALL rd:t2%=TIME-T%
280 PROCinfo("asm 64KB write cs",STR$t1%):PROCinfo("asm 64KB read cs",STR$t2%):PROCinfo("burst errs",STR$(?&70+256*?&71))
290 PROCchk("asm 64KB burst",?&70=0 AND ?&71=0)
300 CALL coh:PROCinfo("coherence errs",STR$(?&70+256*?&71))
310 PROCchk("write then read back 16K",?&70=0 AND ?&71=0)
320 PROCpage(0):PROCend
330 DEF PROCpage(N%):?&FCFD=(N% DIV &10000) AND 255:?&FCFE=(N% DIV 256) AND 255:?&FCFF=N% AND 255:ENDPROC
